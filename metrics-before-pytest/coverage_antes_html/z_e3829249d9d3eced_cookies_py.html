22class RAMCookieJar(QNetworkCookieJar): 

23 

24 """An in-RAM cookie jar. 

25 

26 Signals: 

27 changed: Emitted when the cookie store was changed. 

28 """ 

29 

30 changed = pyqtSignal() 

31 

32 def __repr__(self): 

33 return utils.get_repr(self, count=len(self.allCookies())) 

34 

35 def setCookiesFromUrl(self, cookies, url): 

36 """Add the cookies in the cookies list to this cookie jar. 

37 

38 Args: 

39 cookies: A list of QNetworkCookies. 

40 url: The URL to set the cookies for. 

41 

42 Return: 

43 True if one or more cookies are set for 'url', otherwise False. 

44 """ 

45 accept = config.instance.get('content.cookies.accept', url=url) 

46 

47 if 'log-cookies' in objects.debug_flags: 

48 log.network.debug('Cookie on {} -> applying setting {}' 

49 .format(url.toDisplayString(), accept)) 

50 

51 if accept == 'never': 

52 return False 

53 else: 

54 self.changed.emit() 

55 return super().setCookiesFromUrl(cookies, url)

58class CookieJar(RAMCookieJar): 

59 

60 """A cookie jar saving cookies to disk. 

61 

62 Attributes: 

63 _lineparser: The LineParser managing the cookies file. 

64 """ 

65 

66 def __init__(self, parent=None, *, line_parser=None): 

67 super().__init__(parent) 

68 

69 if line_parser: 

70 self._lineparser = line_parser 

71 else: 

72 self._lineparser = lineparser.LineParser( 

73 standarddir.data(), 'cookies', binary=True, parent=self) 

74 self.parse_cookies() 

75 config.instance.changed.connect(self._on_cookies_store_changed) 

76 objreg.get('save-manager').add_saveable( 

77 'cookies', self.save, self.changed, 

78 config_opt='content.cookies.store') 

79 

80 def parse_cookies(self): 

81 """Parse cookies from lineparser and store them.""" 

82 cookies: Sequence[QNetworkCookie] = [] 

83 for line in self._lineparser: 

84 line_cookies = QNetworkCookie.parseCookies(line) 

85 cookies += line_cookies # type: ignore[operator] 

86 self.setAllCookies(cookies) 

87 

88 def purge_old_cookies(self): 

89 """Purge expired cookies from the cookie jar.""" 

90 # Based on: 

91 # https://doc.qt.io/archives/qt-5.5/qtwebkitexamples-webkitwidgets-browser-cookiejar-cpp.html 

92 now = QDateTime.currentDateTime() 

93 cookies = [c for c in self.allCookies() 

94 if c.isSessionCookie() or 

95 c.expirationDate() >= now] # type: ignore[operator] 

96 self.setAllCookies(cookies) 

97 

98 def save(self): 

99 """Save cookies to disk.""" 

100 self.purge_old_cookies() 

101 lines = [] 

102 for cookie in self.allCookies(): 

103 if not cookie.isSessionCookie(): 

104 lines.append(cookie.toRawForm()) 

105 self._lineparser.data = lines 

106 self._lineparser.save() 

107 

108 @config.change_filter('content.cookies.store') 

109 def _on_cookies_store_changed(self): 

110 """Delete stored cookies if cookies.store changed.""" 

111 if not config.val.content.cookies.store: 

112 self._lineparser.data = [] 

113 self._lineparser.save() 

114 self.changed.emit()

117def init(qapp): 

118 """Initialize the global cookie jars.""" 

119 global cookie_jar, ram_cookie_jar 

120 cookie_jar = CookieJar(qapp) 

121 ram_cookie_jar = RAMCookieJar(qapp)
