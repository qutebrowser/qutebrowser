191class QtWarningFilter(logging.Filter): 

192 

193 """Filter to filter Qt warnings. 

194 

195 Attributes: 

196 _pattern: The start of the message. 

197 """ 

198 

199 def __init__(self, pattern: str) -> None: 

200 super().__init__() 

201 self._pattern = pattern 

202 

203 def filter(self, record: logging.LogRecord) -> bool: 

204 """Determine if the specified record is to be logged.""" 

205 do_log = not record.msg.strip().startswith(self._pattern) 

206 return do_log
