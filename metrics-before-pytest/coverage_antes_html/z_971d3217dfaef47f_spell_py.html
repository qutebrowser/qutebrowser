7"""Installing and configuring spell-checking for QtWebEngine.""" 

8 

9import glob 

10import os 

11import os.path 

12import re 

13 

14from qutebrowser.utils import log, message, standarddir 

15 

16_DICT_VERSION_RE = re.compile(r".+-(?P<version>[0-9]+-[0-9]+?)\.bdic") 

17 

18 

19def version(filename): 

20 """Extract the version number from the dictionary file name.""" 

21 match = _DICT_VERSION_RE.fullmatch(filename) 

22 if match is None: 

23 message.warning( 

24 "Found a dictionary with a malformed name: {}".format(filename)) 

25 return None 

26 return tuple(int(n) for n in match.group('version').split('-')) 

27 

28 

29def dictionary_dir(): 

30 """Return the path (str) to the QtWebEngine's dictionaries directory.""" 

31 return os.path.join(standarddir.data(), 'qtwebengine_dictionaries') 

32 

33 

34def local_files(code): 

35 """Return all installed dictionaries for the given code. 

36 

37 The returned dictionaries are sorted by version, therefore the latest will 

38 be the first element. The list will be empty if no dictionaries are found. 

39 """ 

40 pathname = os.path.join(dictionary_dir(), '{}*.bdic'.format(code)) 

41 matching_dicts = glob.glob(pathname) 

42 versioned_dicts = [] 

43 for matching_dict in matching_dicts: 

44 parsed_version = version(matching_dict) 

45 if parsed_version is not None: 

46 filename = os.path.basename(matching_dict) 

47 log.config.debug('Found file for dict {}: {}' 

48 .format(code, filename)) 

49 versioned_dicts.append((parsed_version, filename)) 

50 return [filename for version, filename 

51 in sorted(versioned_dicts, reverse=True)] 

52 

53 

54def local_filename(code): 

55 """Return the newest installed dictionary for the given code. 

56 

57 Return the filename of the installed dictionary with the highest version 

58 number or None if the dictionary is not installed. 

59 """ 

60 all_installed = local_files(code) 

61 return all_installed[0] if all_installed else None 

62 

63 

64def init(): 

65 """Initialize the dictionary path.""" 

66 dict_dir = dictionary_dir() 

67 os.environ['QTWEBENGINE_DICTIONARIES_PATH'] = dict_dir
