5"""Simplistic ELF parser to get the QtWebEngine/Chromium versions. 

6 

7I know what you must be thinking when reading this: "Why on earth does qutebrowser have 

8an ELF parser?!". For one, because writing one was an interesting learning exercise. But 

9there's actually a reason it's here: QtWebEngine 5.15.x versions come with different 

10underlying Chromium versions, but there is no API to get the version of 

11QtWebEngine/Chromium... 

12 

13We can instead: 

14 

15a) Look at the Qt runtime version (qVersion()). This often doesn't actually correspond 

16to the QtWebEngine version (as that can be older/newer). Since there will be a 

17QtWebEngine 5.15.3 release, but not Qt itself (due to LTS licensing restrictions), this 

18isn't a reliable source of information. 

19 

20b) Look at the PyQtWebEngine version (PyQt5.QtWebEngine.PYQT_WEBENGINE_VERSION_STR). 

21This is a good first guess (especially for our Windows/macOS releases), but still isn't 

22certain. Linux distributions often push a newer QtWebEngine before the corresponding 

23PyQtWebEngine release, and some (*cough* Gentoo *cough*) even publish QtWebEngine 

24"5.15.2" but upgrade the underlying Chromium. 

25 

26c) Parse the user agent. This is what qutebrowser did before this monstrosity was 

27introduced (and still does as a fallback), but for some things (finding the proper 

28commandline arguments to pass) it's too late in the initialization process. 

29 

30d) Spawn QtWebEngine in a subprocess and ask for its user-agent. This takes too long to 

31do it on every startup. 

32 

33e) Ask the package manager for this information. This means we'd need to know (or guess) 

34the package manager and package name. Also see: 

35https://bugs.debian.org/cgi-bin/bugreport.cgi?bug=752114 

36 

37Because of all those issues, we instead look for the (fixed!) version string as part of 

38the user agent header. Because libQt5WebEngineCore is rather big (~120 MB), we don't 

39want to search through the entire file, so we instead have a simplistic ELF parser here 

40to find the .rodata section. This way, searching the version gets faster by some orders 

41of magnitudes (a couple of us instead of ms). 

42 

43This is a "best effort" parser. If it errors out, we instead end up relying on the 

44PyQtWebEngine version, which is the next best thing. 

45"""

177def get_rodata_header(f: IO[bytes]) -> SectionHeader: 

178 """Parse an ELF file and find the .rodata section header.""" 

179 ident = Ident.parse(f) 

180 if ident.magic != b'\x7fELF': 180 ↛ 181line 180 didn't jump to line 181 because the condition on line 180 was never true

181 raise binparsing.ParseError(f"Invalid magic {ident.magic!r}") 

182 

183 if ident.data != Endianness.little: 183 ↛ 184line 183 didn't jump to line 184 because the condition on line 183 was never true

184 raise binparsing.ParseError("Big endian is unsupported") 

185 

186 if ident.version != 1: 186 ↛ 187line 186 didn't jump to line 187 because the condition on line 186 was never true

187 raise binparsing.ParseError(f"Only version 1 is supported, not {ident.version}") 

188 

189 header = Header.parse(f, bitness=ident.klass) 

190 

191 # Read string table 

192 binparsing.safe_seek(f, header.shoff + header.shstrndx * header.shentsize) 

193 shstr = SectionHeader.parse(f, bitness=ident.klass) 

194 

195 binparsing.safe_seek(f, shstr.offset) 

196 string_table = binparsing.safe_read(f, shstr.size) 

197 

198 # Back to all sections 

199 for i in range(header.shnum): 199 ↛ 200line 199 didn't jump to line 200 because the loop on line 199 never started

200 binparsing.safe_seek(f, header.shoff + i * header.shentsize) 

201 sh = SectionHeader.parse(f, bitness=ident.klass) 

202 name = string_table[sh.name:].split(b'\x00')[0] 

203 if name == b'.rodata': 

204 return sh 

205 

206 raise binparsing.ParseError("No .rodata section found")

218def _find_versions(data: bytes) -> Versions: 

219 """Find the version numbers in the given data. 

220 

221 Note that 'data' can actually be a mmap.mmap, but typing doesn't handle that 

222 correctly: https://github.com/python/typeshed/issues/1467 

223 """ 

224 pattern = br'\x00QtWebEngine/([0-9.]+) Chrome/([0-9.]+)\x00' 

225 match = re.search(pattern, data) 

226 if match is not None: 

227 try: 

228 return Versions( 

229 webengine=match.group(1).decode('ascii'), 

230 chromium=match.group(2).decode('ascii'), 

231 ) 

232 except UnicodeDecodeError as e: 

233 raise binparsing.ParseError(e) 

234 

235 # Here it gets even more crazy: Sometimes, we don't have the full UA in one piece 

236 # in the string table somehow (?!). However, Qt 6.2 added a separate 

237 # qWebEngineChromiumVersion(), with PyQt wrappers following later. And *that* 

238 # apparently stores the full version in the string table separately from the UA. 

239 # As we clearly didn't have enough crazy heuristics yet so far, let's hunt for it! 

240 

241 # We first get the partial Chromium version from the UA: 

242 match = re.search(pattern[:-4], data) # without trailing literal \x00 

243 if match is None: 

244 raise binparsing.ParseError("No match in .rodata") 

245 

246 webengine_bytes = match.group(1) 

247 partial_chromium_bytes = match.group(2) 

248 if b"." not in partial_chromium_bytes or len(partial_chromium_bytes) < 6: 

249 # some sanity checking 

250 raise binparsing.ParseError("Inconclusive partial Chromium bytes") 

251 

252 # And then try to find the *full* string, stored separately, based on the 

253 # partial one we got above. 

254 pattern = br"\x00(" + re.escape(partial_chromium_bytes) + br"[0-9.]+)\x00" 

255 match = re.search(pattern, data) 

256 if match is None: 

257 raise binparsing.ParseError("No match in .rodata for full version") 

258 

259 chromium_bytes = match.group(1) 

260 try: 

261 return Versions( 

262 webengine=webengine_bytes.decode('ascii'), 

263 chromium=chromium_bytes.decode('ascii'), 

264 ) 

265 except UnicodeDecodeError as e: 

266 raise binparsing.ParseError(e)

269def _parse_from_file(f: IO[bytes]) -> Versions: 

270 """Parse the ELF file from the given path.""" 

271 sh = get_rodata_header(f) 

272 

273 rest = sh.offset % mmap.ALLOCATIONGRANULARITY 

274 mmap_offset = sh.offset - rest 

275 mmap_size = sh.size + rest 

276 

277 try: 

278 with mmap.mmap( 

279 f.fileno(), 

280 mmap_size, 

281 offset=mmap_offset, 

282 access=mmap.ACCESS_READ, 

283 ) as mmap_data: 

284 return _find_versions(cast(bytes, mmap_data)) 

285 except (OSError, OverflowError) as e: 

286 log.misc.debug(f"mmap failed ({e}), falling back to reading", exc_info=True) 

287 binparsing.safe_seek(f, sh.offset) 

288 data = binparsing.safe_read(f, sh.size) 

289 return _find_versions(data)
