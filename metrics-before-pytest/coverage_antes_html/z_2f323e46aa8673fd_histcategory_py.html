17class HistoryCategory(QSqlQueryModel, BaseCategory): 

18 

19 """A completion category that queries the SQL history store.""" 

20 

21 def __init__(self, *, database: sql.Database, 

22 delete_func: util.DeleteFuncType | None = None, 

23 parent: QWidget | None = None) -> None: 

24 """Create a new History completion category.""" 

25 super().__init__(parent=parent) 

26 self._database = database 

27 self.name = "History" 

28 self._query: sql.Query | None = None 

29 

30 # advertise that this model filters by URL and title 

31 self.columns_to_filter = [0, 1] 

32 self.delete_func = delete_func 

33 self._empty_prefix: str | None = None 

34 

35 def _atime_expr(self): 

36 """If max_items is set, return an expression to limit the query.""" 

37 max_items = config.val.completion.web_history.max_items 

38 # HistoryCategory should not be added to the completion in that case. 

39 assert max_items != 0 

40 

41 if max_items < 0: 

42 return '' 

43 

44 min_atime = self._database.query(' '.join([ 

45 'SELECT min(last_atime) FROM', 

46 '(SELECT last_atime FROM CompletionHistory', 

47 'ORDER BY last_atime DESC LIMIT :limit)', 

48 ])).run(limit=max_items).value() 

49 

50 if not min_atime: 

51 # if there are no history items, min_atime may be '' (issue #2849) 

52 return '' 

53 

54 return "AND last_atime >= {}".format(min_atime) 

55 

56 def set_pattern(self, pattern): 

57 """Set the pattern used to filter results. 

58 

59 Args: 

60 pattern: string pattern to filter by. 

61 """ 

62 raw_pattern = pattern 

63 if (self._empty_prefix is not None and raw_pattern.startswith( 

64 self._empty_prefix)): 

65 log.sql.debug('Skipping query on {} due to ' 

66 'prefix {} returning nothing.' 

67 .format(raw_pattern, self._empty_prefix)) 

68 return 

69 self._empty_prefix = None 

70 

71 # escape to treat a user input % or _ as a literal, not a wildcard 

72 pattern = pattern.replace('%', '\\%') 

73 pattern = pattern.replace('_', '\\_') 

74 words = ['%{}%'.format(w) for w in pattern.split(' ')] 

75 

76 # build a where clause to match all of the words in any order 

77 # given the search term "a b", the WHERE clause would be: 

78 # (url LIKE '%a%' OR title LIKE '%a%') AND 

79 # (url LIKE '%b%' OR title LIKE '%b%') 

80 where_clause = ' AND '.join( 

81 "(url LIKE :{val} escape '\\' OR title LIKE :{val} escape '\\')" 

82 .format(val=i) for i in range(len(words))) 

83 

84 # replace ' in timestamp-format to avoid breaking the query 

85 timestamp_format = config.val.completion.timestamp_format or '' 

86 timefmt = ("strftime('{}', last_atime, 'unixepoch', 'localtime')" 

87 .format(timestamp_format.replace("'", "`"))) 

88 

89 try: 

90 if (not self._query or 

91 len(words) != len(self._query.bound_values())): 

92 # if the number of words changed, we need to generate a new 

93 # query otherwise, we can reuse the prepared query for 

94 # performance 

95 self._query = self._database.query(' '.join([ 

96 "SELECT url, title, {}".format(timefmt), 

97 "FROM CompletionHistory", 

98 # the incoming pattern will have literal % and _ escaped we 

99 # need to tell SQL to treat '\' as an escape character 

100 'WHERE ({})'.format(where_clause), 

101 self._atime_expr(), 

102 "ORDER BY last_atime DESC", 

103 ]), forward_only=False) 

104 

105 with debug.log_time('sql', 'Running completion query'): 

106 self._query.run(**{ 

107 str(i): w for i, w in enumerate(words)}) 

108 except sql.KnownError as e: 

109 # Sometimes, the query we built up was invalid, for example, 

110 # due to a large amount of words. 

111 # Also catches failures in the DB we can't solve. 

112 message.error("Error with SQL query: {}".format(e.text())) 

113 return 

114 self.setQuery(self._query.query) 

115 if not self.rowCount() and not self.canFetchMore(): 

116 self._empty_prefix = raw_pattern 

117 

118 def removeRows(self, row, _count, _parent=None): 

119 """Override QAbstractItemModel::removeRows to re-run SQL query.""" 

120 # re-run query to reload updated table 

121 assert self._query is not None 

122 with debug.log_time('sql', 'Re-running completion query post-delete'): 

123 self._query.run() 

124 self.setQuery(self._query.query) 

125 while self.rowCount() < row: 

126 self.fetchMore() 

127 return True
