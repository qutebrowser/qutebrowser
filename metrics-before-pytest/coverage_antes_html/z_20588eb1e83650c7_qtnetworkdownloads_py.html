33class DownloadItem(downloads.AbstractDownloadItem): 

34 

35 """A single download currently running. 

36 

37 There are multiple ways the data can flow from the QNetworkReply to the 

38 disk. 

39 

40 If the filename/file object is known immediately when starting the 

41 download, QNetworkReply's readyRead writes to the target file directly. 

42 

43 If not, readyRead is ignored and with self._read_timer we periodically read 

44 into the self._buffer BytesIO slowly, so some broken servers don't close 

45 our connection. 

46 

47 As soon as we know the file object, we copy self._buffer over and the next 

48 readyRead will write to the real file object. 

49 

50 Attributes: 

51 _retry_info: A _RetryInfo instance. 

52 _buffer: A BytesIO object to buffer incoming data until we know the 

53 target file. 

54 _read_timer: A Timer which reads the QNetworkReply into self._buffer 

55 periodically. 

56 _reply: The QNetworkReply associated with this download. 

57 _autoclose: Whether to close the associated file when the download is 

58 done. 

59 

60 Signals: 

61 adopt_download: Emitted when a download is retried and should be 

62 adopted by the QNAM if needed. 

63 arg 0: The new DownloadItem 

64 """ 

65 

66 adopt_download = pyqtSignal(object) # DownloadItem

68 def __init__(self, reply, manager): 

69 """Constructor. 

70 

71 Args: 

72 reply: The QNetworkReply to download. 

73 """ 

74 super().__init__(manager=manager, parent=manager) 

75 self.fileobj: IO[bytes] | None = None 

76 self.raw_headers: dict[bytes, bytes] = {}

83 self._read_timer.setInterval(500) 

84 self._read_timer.timeout.connect(self._on_read_timer_timeout) 

85 self._url = reply.url() 

86 self._init_reply(reply)

88 def _create_fileobj(self): 

89 """Create a file object using the internal filename.""" 

90 assert self._filename is not None 

91 try: 

92 # pylint: disable=consider-using-with 

93 fileobj = open(self._filename, 'wb') 

94 except OSError as e: 

95 self._die(e.strerror) 

96 else: 

97 self._set_fileobj(fileobj)

99 def _do_die(self): 

100 """Abort the download and emit an error.""" 

101 self._read_timer.stop() 

102 if self._reply is None: 

103 log.downloads.debug("Reply gone while dying") 

104 return 

105 

106 self._reply.downloadProgress.disconnect() 

107 self._reply.finished.disconnect() 

108 self._reply.errorOccurred.disconnect() 

109 self._reply.readyRead.disconnect() 

110 

111 with qtlog.hide_qt_warning('QNetworkReplyImplPrivate::error: Internal ' 

112 'problem, this method must only be called ' 

113 'once.'): 

114 # See https://codereview.qt-project.org/#/c/107863/ 

115 self._reply.abort() 

116 self._reply.deleteLater() 

117 self._reply = None 

118 if self.fileobj is not None: 

119 pos = self.fileobj.tell() 

120 log.downloads.debug(f"File position at error: {pos}") 

121 try: 

122 self.fileobj.close() 

123 except OSError: 

124 log.downloads.exception("Error while closing file object") 

125 

126 if pos == 0: 

127 # Empty remaining file 

128 filename = self._get_open_filename() 

129 log.downloads.debug(f"Removing empty file at {filename}") 

130 try: 

131 os.remove(filename) 

132 except OSError: 

133 log.downloads.exception("Error while removing empty file")

135 def _init_reply(self, reply): 

136 """Set a new reply and connect its signals. 

137 

138 Args: 

139 reply: The QNetworkReply to handle. 

140 """ 

141 self.done = False 

142 self.successful = False 

143 self._reply = reply 

144 reply.setReadBufferSize(16 * 1024 * 1024) # 16 MB 

145 

146 reply.downloadProgress.connect(self.stats.on_download_progress) 

147 reply.finished.connect(self._on_reply_finished) 

148 reply.errorOccurred.connect(self._on_reply_error) 

149 reply.readyRead.connect(self._on_ready_read) 

150 reply.metaDataChanged.connect(self._on_meta_data_changed) 

151 reply.redirected.connect(self._on_redirected) 

152 

153 self._retry_info = _RetryInfo(request=reply.request(), 

154 manager=reply.manager()) 

155 if not self.fileobj: 

156 self._read_timer.start() 

157 # We could have got signals before we connected slots to them. 

158 # Here no signals are connected to the DownloadItem yet, so we use a 

159 # singleShot QTimer to emit them after they are connected. 

160 if reply.error() != QNetworkReply.NetworkError.NoError: 

161 QTimer.singleShot(0, lambda: self._die(reply.errorString()))

163 @pyqtSlot(QUrl) 

164 def _on_redirected(self, url): 

165 if self._reply is None: 

166 log.downloads.warning(f"redirected: REPLY GONE -> {url}") 

167 else: 

168 log.downloads.debug(f"redirected: {self._reply.url()} -> {url}")

170 def _do_cancel(self): 

171 self._read_timer.stop() 

172 if self._reply is not None: 

173 self._reply.finished.disconnect(self._on_reply_finished) 

174 self._reply.abort() 

175 self._reply.deleteLater() 

176 self._reply = None 

177 if self.fileobj is not None: 

178 self.fileobj.close() 

179 self.cancelled.emit()

181 @pyqtSlot() 

182 def retry(self): 

183 """Retry a failed download.""" 

184 assert self.done 

185 assert not self.successful 

186 assert self._retry_info is not None 

187 

188 # Not calling self.cancel() here because the download is done (albeit 

189 # unsuccessfully) 

190 self.remove() 

191 self.delete() 

192 

193 new_reply = self._retry_info.manager.get(self._retry_info.request) 

194 new_download = self._manager.fetch(new_reply, 

195 suggested_filename=self.basename) 

196 self.adopt_download.emit(new_download)

249 def _set_fileobj(self, fileobj, *, autoclose=True): 

250 """Set the file object to write the download to. 

251 

252 Args: 

253 fileobj: A file-like object. 

254 """ 

255 assert self._reply is not None 

256 if self.fileobj is not None: # pragma: no cover 

257 raise ValueError("fileobj was already set! Old: {}, new: " 

258 "{}".format(self.fileobj, fileobj)) 

259 self.fileobj = fileobj 

260 self._autoclose = autoclose 

261 try: 

262 self._read_timer.stop() 

263 log.downloads.debug("buffer: {} bytes".format(self._buffer.tell())) 

264 self._buffer.seek(0) 

265 shutil.copyfileobj(self._buffer, fileobj) 

266 self._buffer.close() 

267 if self._reply.isFinished(): 

268 # Downloading to the buffer in RAM has already finished so we 

269 # write out the data and clean up now. 

270 self._on_reply_finished() 

271 else: 

272 # Since the buffer already might be full, on_ready_read might 

273 # not be called at all anymore, so we force it here to flush 

274 # the buffer and continue receiving new data. 

275 self._on_ready_read() 

276 except OSError as e: 

277 self._die(e.strerror)

280 self._set_fileobj(fileobj)

282 def _finish_download(self): 

283 """Write buffered data to disk and finish the QNetworkReply.""" 

284 assert self._reply is not None 

285 assert self.fileobj is not None 

286 log.downloads.debug("Finishing download...") 

287 if self._reply.isOpen(): 

288 self.fileobj.write(self._reply.readAll()) 

289 if self._autoclose: 

290 self.fileobj.close() 

291 self.successful = self._reply.error() == QNetworkReply.NetworkError.NoError 

292 self._reply.close() 

293 self._reply.deleteLater() 

294 self._reply = None 

295 self.finished.emit() 

296 self.done = True 

297 log.downloads.debug("Download {} finished".format(self.basename)) 

298 self.data_changed.emit()

300 @pyqtSlot() 

301 def _on_reply_finished(self): 

302 """Clean up when the download was finished. 

303 

304 Note when this gets called, only the QNetworkReply has finished. This 

305 doesn't mean the download (i.e. writing data to the disk) is finished 

306 as well. Therefore, we can't close() the QNetworkReply in here yet. 

307 """ 

308 if self._reply is None: 

309 return 

310 self._read_timer.stop() 

311 self.stats.finish() 

312 log.downloads.debug("Reply finished, fileobj {}".format(self.fileobj)) 

313 if self.fileobj is not None: 

314 # We can do a "delayed" write immediately to empty the buffer and 

315 # clean up. 

316 self._finish_download() 

317 

318 @pyqtSlot() 

319 def _on_ready_read(self): 

320 """Read available data and save file when ready to read.""" 

321 if self.fileobj is None or self._reply is None: 

322 # No filename has been set yet (so we don't empty the buffer) or we 

323 # got a readyRead after the reply was finished (which happens on 

324 # qute://log for example). 

325 return 

326 if not self._reply.isOpen(): 

327 raise OSError("Reply is closed!") 

328 try: 

329 self.fileobj.write(self._reply.readAll()) 

330 except OSError as e: 

331 self._die(e.strerror) 

332 

333 @pyqtSlot('QNetworkReply::NetworkError') 

334 def _on_reply_error(self, code): 

335 """Handle QNetworkReply errors.""" 

336 if code == QNetworkReply.NetworkError.OperationCanceledError: 

337 return 

338 

339 if self._reply is None: 

340 error = "Unknown error: {}".format( 

341 debug.qenum_key(QNetworkReply, code)) 

342 else: 

343 error = self._reply.errorString() 

344 

345 self._die(error) 

346 

347 @pyqtSlot() 

348 def _on_read_timer_timeout(self): 

349 """Read some bytes from the QNetworkReply periodically.""" 

350 assert self._reply is not None 

351 if not self._reply.isOpen(): 

352 raise OSError("Reply is closed!") 

353 data = self._reply.read(1024) 

354 if data is not None: 

355 self._buffer.write(data) 

356 

357 @pyqtSlot() 

358 def _on_meta_data_changed(self): 

359 """Update the download's metadata.""" 

360 if self._reply is None: 

361 return 

362 self.raw_headers = {} 

363 for key, value in self._reply.rawHeaderPairs(): 

364 self.raw_headers[bytes(key)] = bytes(value)

366 def _uses_nam(self, nam): 

367 """Check if this download uses the given QNetworkAccessManager.""" 

368 assert self._retry_info is not None 

369 running_nam = self._reply is not None and self._reply.manager() is nam 

370 # user could request retry after tab is closed. 

371 retry_nam = (self.done and (not self.successful) and 

372 self._retry_info.manager is nam) 

373 return running_nam or retry_nam

376class DownloadManager(downloads.AbstractDownloadManager): 

377 

378 """Manager for currently running downloads. 

379 

380 Attributes: 

381 _networkmanager: A NetworkManager for generic downloads. 

382 

383 Class attributes: 

384 _MAX_REDIRECTS: The maximum redirection count. 

385 """ 

386 

387 # Same as many browsers 

388 # https://fetch.spec.whatwg.org/#http-redirect-fetch 

389 # https://source.chromium.org/chromium/chromium/src/+/main:net/url_request/url_request.h;l=97;drc=3c19a2edb96d3d5b56a7481349a357fdbdf8ecf0 

390 # https://stackoverflow.com/questions/9384474/in-chrome-how-many-redirects-are-too-many 

391 _MAX_REDIRECTS = 20 

392 

393 def __init__(self, parent=None): 

394 super().__init__(parent) 

395 self._networkmanager = networkmanager.NetworkManager( 

396 win_id=None, tab_id=None, 

397 private=config.val.content.private_browsing, parent=self) 

398 

399 @pyqtSlot('QUrl') 

400 def get(self, url, cache=True, **kwargs): 

401 """Start a download with a link URL. 

402 

403 Args: 

404 url: The URL to get, as QUrl 

405 cache: If set to False, don't cache the response. 

406 **kwargs: passed to get_request(). 

407 

408 Return: 

409 The created DownloadItem. 

410 """ 

411 if not url.isValid(): 

412 urlutils.invalid_url_error(url, "start download") 

413 return None 

414 

415 req = QNetworkRequest(url) 

416 

417 user_agent = websettings.user_agent(url) 

418 req.setHeader(QNetworkRequest.KnownHeaders.UserAgentHeader, user_agent) 

419 if not cache: 

420 req.setAttribute(QNetworkRequest.Attribute.CacheSaveControlAttribute, False) 

421 

422 # Needed for Qt 5, default on Qt 6 

423 # We don't set this on the QNAM because QtWebKit handles redirects manually. 

424 req.setAttribute( 

425 QNetworkRequest.Attribute.RedirectPolicyAttribute, 

426 QNetworkRequest.RedirectPolicy.NoLessSafeRedirectPolicy, 

427 ) 

428 req.setMaximumRedirectsAllowed(self._MAX_REDIRECTS) 

429 

430 return self.get_request(req, **kwargs)

438 mhtml.start_download_checked(target, tab=tab)

454 def _get_suggested_filename(self, request): 

455 """Get the suggested filename for the given request.""" 

456 filename_url = request.url() 

457 if request.url().scheme().lower() == 'data': 

458 # We might be downloading a binary blob embedded on a page or even 

459 # generated dynamically via javascript. If we happen to know where it's 

460 # coming from, we can try to figure out a more sensible name than the base64 

461 # content of the data. 

462 origin = request.originatingObject() 

463 try: 

464 filename_url = origin.url() 

465 except AttributeError: 

466 # Raised either if origin is None or some object that doesn't 

467 # have its own url. We're probably fine with a default fallback 

468 # based on the data URL then. 

469 pass 

470 

471 return urlutils.filename_from_url(filename_url, fallback='qutebrowser-download')

473 def get_request(self, request, *, target=None, 

474 suggested_fn=None, **kwargs): 

475 """Start a download with a QNetworkRequest. 

476 

477 Args: 

478 request: The QNetworkRequest to download. 

479 target: Where to save the download as downloads.DownloadTarget. 

480 suggested_fn: The filename to use for the file. 

481 **kwargs: Passed to _fetch_request. 

482 

483 Return: 

484 The created DownloadItem. 

485 """ 

486 # WORKAROUND for Qt corrupting data loaded from cache: 

487 # https://bugreports.qt.io/browse/QTBUG-42757 

488 request.setAttribute(QNetworkRequest.Attribute.CacheLoadControlAttribute, 

489 QNetworkRequest.CacheLoadControl.AlwaysNetwork) 

490 

491 if suggested_fn is None: 

492 suggested_fn = self._get_suggested_filename(request) 

493 

494 return self._fetch_request(request, 

495 target=target, 

496 suggested_filename=suggested_fn, 

497 **kwargs)

499 def _fetch_request(self, request, *, qnam=None, **kwargs): 

500 """Download a QNetworkRequest to disk. 

501 

502 Args: 

503 request: The QNetworkRequest to download. 

504 qnam: The QNetworkAccessManager to use. 

505 **kwargs: passed to fetch(). 

506 

507 Return: 

508 The created DownloadItem. 

509 """ 

510 if qnam is None: 

511 qnam = self._networkmanager 

512 reply = qnam.get(request) 

513 return self.fetch(reply, **kwargs)

515 @pyqtSlot('QNetworkReply') 

516 def fetch(self, reply, *, target=None, auto_remove=False, 

517 suggested_filename=None, prompt_download_directory=None): 

518 """Download a QNetworkReply to disk. 

519 

520 Args: 

521 reply: The QNetworkReply to download. 

522 target: Where to save the download as downloads.DownloadTarget. 

523 auto_remove: Whether to remove the download even if 

524 downloads.remove_finished is set to -1. 

525 suggested_filename: The filename to use for the file. 

526 prompt_download_directory: Whether to prompt for a location to 

527 download the file to. 

528 

529 Return: 

530 The created DownloadItem. 

531 """ 

532 if not suggested_filename: 

533 try: 

534 suggested_filename = target.suggested_filename() 

535 except downloads.NoFilenameError: 

536 _, suggested_filename = httpheaders.parse_content_disposition(reply) 

537 log.downloads.debug("fetch: {} -> {}".format(reply.url(), 

538 suggested_filename)) 

539 download = DownloadItem(reply, manager=self) 

540 self._init_item(download, auto_remove, suggested_filename) 

541 

542 if download.cancel_for_origin(): 

543 return download 

544 

545 if target is not None: 

546 download.set_target(target) 

547 return download 

548 

549 # Neither filename nor fileobj were given 

550 

551 filename = downloads.immediate_download_path(prompt_download_directory) 

552 if filename is not None: 

553 # User doesn't want to be asked, so just use the download_dir 

554 target = downloads.FileDownloadTarget(filename) 

555 download.set_target(target) 

556 return download 

557 

558 # Ask the user for a filename 

559 question = downloads.get_filename_question( 

560 suggested_filename=suggested_filename, url=reply.url(), 

561 parent=self) 

562 self._init_filename_question(question, download) 

563 message.global_bridge.ask(question, blocking=False) 

564 

565 return download

567 def has_downloads_with_nam(self, nam): 

568 """Check if the DownloadManager has any downloads with the given QNAM. 

569 

570 Args: 

571 nam: The QNetworkAccessManager to check. 

572 

573 Return: 

574 A boolean. 

575 """ 

576 assert nam.adopted_downloads == 0 

577 for download in self.downloads: 

578 assert isinstance(download, DownloadItem), download 

579 if download._uses_nam(nam): # pylint: disable=protected-access 

580 nam.adopt_download(download) 

581 return nam.adopted_downloads
