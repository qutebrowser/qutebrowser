24class ConsoleLineEdit(miscwidgets.CommandLineEdit): 

25 

26 """A QLineEdit which executes entered code and provides a history. 

27 

28 Attributes: 

29 _history: The command history of executed commands. 

30 

31 Signals: 

32 execute: Emitted when a commandline should be executed. 

33 """ 

34 

35 execute = pyqtSignal(str) 

36 

37 def __init__(self, _namespace, parent): 

38 """Constructor. 

39 

40 Args: 

41 _namespace: The local namespace of the interpreter. 

42 """ 

43 super().__init__(parent=parent) 

44 self._history = cmdhistory.History(parent=self) 

45 self.returnPressed.connect(self.on_return_pressed) 

46 

47 @pyqtSlot() 

48 def on_return_pressed(self): 

49 """Execute the line of code which was entered.""" 

50 self._history.stop() 

51 text = self.text() 

52 if text: 

53 self._history.append(text) 

54 self.execute.emit(text) 

55 self.setText('') 

56 

57 def history_prev(self): 

58 """Go back in the history.""" 

59 try: 

60 if not self._history.is_browsing(): 

61 item = self._history.start(self.text().strip()) 

62 else: 

63 item = self._history.previtem() 

64 except (cmdhistory.HistoryEmptyError, 

65 cmdhistory.HistoryEndReachedError): 

66 return 

67 self.setText(item) 

68 

69 def history_next(self): 

70 """Go forward in the history.""" 

71 if not self._history.is_browsing(): 

72 return 

73 try: 

74 item = self._history.nextitem() 

75 except cmdhistory.HistoryEndReachedError: 

76 return 

77 self.setText(item) 

78 

79 def keyPressEvent(self, e): 

80 """Override keyPressEvent to handle special keypresses.""" 

81 if e.key() == Qt.Key.Key_Up: 

82 self.history_prev() 

83 e.accept() 

84 elif e.key() == Qt.Key.Key_Down: 

85 self.history_next() 

86 e.accept() 

87 elif e.modifiers() & Qt.KeyboardModifier.ControlModifier and e.key() == Qt.Key.Key_C: 

88 self.setText('') 

89 e.accept() 

90 else: 

91 super().keyPressEvent(e)

94class ConsoleTextEdit(QTextEdit): 

95 

96 """Custom QTextEdit for console output.""" 

97 

98 def __init__(self, parent=None): 

99 super().__init__(parent) 

100 self.setAcceptRichText(False) 

101 self.setReadOnly(True) 

102 self.setFocusPolicy(Qt.FocusPolicy.ClickFocus) 

103 

104 def __repr__(self): 

105 return utils.get_repr(self) 

106 

107 def append_text(self, text): 

108 """Append new text and scroll output to bottom. 

109 

110 We can't use Qt's way to append stuff because that inserts weird 

111 newlines. 

112 """ 

113 self.moveCursor(QTextCursor.MoveOperation.End) 

114 self.insertPlainText(text) 

115 scrollbar = self.verticalScrollBar() 

116 assert scrollbar is not None 

117 scrollbar.setValue(scrollbar.maximum())

120class ConsoleWidget(QWidget): 

121 

122 """A widget with an interactive Python console. 

123 

124 Attributes: 

125 _lineedit: The line edit in the console. 

126 _output: The output widget in the console. 

127 _vbox: The layout which contains everything. 

128 _more: A flag which is set when more input is expected. 

129 _buffer: The buffer for multi-line commands. 

130 _interpreter: The InteractiveInterpreter to execute code with. 

131 """ 

132 

133 STYLESHEET = """ 

134 ConsoleWidget > ConsoleTextEdit, ConsoleWidget > ConsoleLineEdit { 

135 font: {{ conf.fonts.debug_console }}; 

136 } 

137 """ 

138 

139 def __init__(self, parent=None): 

140 super().__init__(parent) 

141 if not hasattr(sys, 'ps1'): 

142 sys.ps1 = '>>> ' 

143 if not hasattr(sys, 'ps2'): 

144 sys.ps2 = '... ' 

145 namespace = { 

146 '__name__': '__console__', 

147 '__doc__': None, 

148 'q_app': QApplication.instance(), 

149 # We use parent as self here because the user "feels" the whole 

150 # console, not just the line edit. 

151 'self': parent, 

152 'objreg': objreg, 

153 } 

154 self._more = False 

155 self._buffer: MutableSequence[str] = [] 

156 self._lineedit = ConsoleLineEdit(namespace, self) 

157 self._lineedit.execute.connect(self.push) 

158 self._output = ConsoleTextEdit() 

159 self.write(self._curprompt()) 

160 self._vbox = QVBoxLayout() 

161 self._vbox.setSpacing(0) 

162 self._vbox.addWidget(self._output) 

163 self._vbox.addWidget(self._lineedit) 

164 stylesheet.set_register(self) 

165 self.setLayout(self._vbox) 

166 self._lineedit.setFocus() 

167 self._interpreter = code.InteractiveInterpreter(namespace) 

168 

169 def __repr__(self): 

170 return utils.get_repr(self, visible=self.isVisible()) 

171 

172 def write(self, line): 

173 """Write a line of text (without added newline) to the output.""" 

174 self._output.append_text(line) 

175 

176 @pyqtSlot(str) 

177 def push(self, line): 

178 """Push a line to the interpreter.""" 

179 self._buffer.append(line) 

180 source = '\n'.join(self._buffer) 

181 self.write(line + '\n') 

182 # We do two special things with the context managers here: 

183 # - We replace stdout/stderr to capture output. Even if we could 

184 # override InteractiveInterpreter's write method, most things are 

185 # printed elsewhere (e.g. by exec). Other Python GUI shells do the 

186 # same. 

187 # - We disable our exception hook, so exceptions from the console get 

188 # printed and don't open a crashdialog. 

189 with utils.fake_io(self.write), utils.disabled_excepthook(): 

190 self._more = self._interpreter.runsource(source, '<console>') 

191 self.write(self._curprompt()) 

192 if not self._more: 

193 self._buffer = [] 

194 

195 def _curprompt(self): 

196 """Get the prompt which is visible currently.""" 

197 return sys.ps2 if self._more else sys.ps1
