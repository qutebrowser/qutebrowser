110 if reverse: 

111 flags |= QWebPage.FindFlag.FindBackward

114 def _call_cb(self, callback, found, text, flags, caller): 

115 """Call the given callback if it's non-None. 

116 

117 Delays the call via a QTimer so the website is re-rendered in between. 

118 

119 Args: 

120 callback: What to call 

121 found: If the text was found 

122 text: The text searched for 

123 flags: The flags searched with 

124 caller: Name of the caller. 

125 """ 

126 found_text = 'found' if found else "didn't find" 

127 # Removing FindWrapsAroundDocument to get the same logging as with 

128 # QtWebEngine 

129 debug_flags = debug.qflags_key( 

130 QWebPage, flags & ~QWebPage.FindFlag.FindWrapsAroundDocument, 

131 klass=QWebPage.FindFlag) 

132 if debug_flags != '0x0000': 

133 flag_text = 'with flags {}'.format(debug_flags) 

134 else: 

135 flag_text = '' 

136 log.webview.debug(' '.join([caller, found_text, text, flag_text]) 

137 .strip()) 

138 if callback is not None: 

139 if caller in ["prev_result", "next_result"]: 

140 if found: 

141 # no wrapping detection 

142 cb_value = browsertab.SearchNavigationResult.found 

143 elif flags & QWebPage.FindBackward: 

144 cb_value = browsertab.SearchNavigationResult.wrap_prevented_top 

145 else: 

146 cb_value = browsertab.SearchNavigationResult.wrap_prevented_bottom 

147 elif caller == "search": 

148 cb_value = found 

149 else: 

150 raise utils.Unreachable(caller) 

151 QTimer.singleShot(0, functools.partial(callback, cb_value)) 

152 

153 self.finished.emit(found)

155 def clear(self): 

156 if self.search_displayed: 

157 self.cleared.emit() 

158 self.search_displayed = False 

159 # We first clear the marked text, then the highlights 

160 self._widget.findText('') 

161 self._widget.findText( 

162 '', QWebPage.FindFlag.HighlightAllOccurrences) # type: ignore[arg-type]

164 def search(self, text, *, ignore_case=usertypes.IgnoreCase.never, 

165 reverse=False, result_cb=None): 

166 # Don't go to next entry on duplicate search 

167 if self.text == text and self.search_displayed: 

168 log.webview.debug("Ignoring duplicate search request" 

169 " for {}, but resetting flags".format(text)) 

170 self._flags = self._args_to_flags(reverse, ignore_case) 

171 return 

172 

173 # Clear old search results, this is done automatically on QtWebEngine. 

174 self.clear() 

175 

176 self.text = text 

177 self.search_displayed = True 

178 self._flags = self._args_to_flags(reverse, ignore_case) 

179 # We actually search *twice* - once to highlight everything, then again 

180 # to get a mark so we can navigate. 

181 found = self._widget.findText(text, self._flags) 

182 self._widget.findText(text, 

183 self._flags | QWebPage.FindFlag.HighlightAllOccurrences) 

184 self._call_cb(result_cb, found, text, self._flags, 'search')

186 def next_result(self, *, wrap=False, callback=None): 

187 self.search_displayed = True 

188 # The int() here makes sure we get a copy of the flags. 

189 flags = QWebPage.FindFlags( 

190 int(self._flags)) # type: ignore[call-overload] 

191 

192 if wrap: 

193 flags |= QWebPage.FindFlag.FindWrapsAroundDocument 

194 

195 found = self._widget.findText(self.text, flags) # type: ignore[arg-type] 

196 self._call_cb(callback, found, self.text, flags, 'next_result') 

197 

198 def prev_result(self, *, wrap=False, callback=None): 

199 self.search_displayed = True 

200 # The int() here makes sure we get a copy of the flags. 

201 flags = QWebPage.FindFlags( 

202 int(self._flags)) # type: ignore[call-overload] 

203 

204 if flags & QWebPage.FindFlag.FindBackward: 

205 flags &= ~QWebPage.FindFlag.FindBackward 

206 else: 

207 flags |= QWebPage.FindFlag.FindBackward 

208 

209 if wrap: 

210 flags |= QWebPage.FindFlag.FindWrapsAroundDocument 

211 

212 found = self._widget.findText(self.text, flags) # type: ignore[arg-type] 

213 self._call_cb(callback, found, self.text, flags, 'prev_result')

729 def find_css(self, selector, callback, error_cb, *, only_visible=False): 

730 utils.unused(error_cb) 

731 mainframe = self._widget.page().mainFrame() 

732 if mainframe is None: 

733 raise browsertab.WebTabError("No frame focused!") 

734 

735 elems = [] 

736 frames = webkitelem.get_child_frames(mainframe) 

737 for f in frames: 

738 frame_elems = cast(Iterable[QWebElement], f.findAllElements(selector)) 

739 for elem in frame_elems: 

740 elems.append(webkitelem.WebKitElement(elem, tab=self._tab)) 

741 

742 if only_visible: 

743 # pylint: disable=protected-access 

744 elems = [e for e in elems if e._is_visible(mainframe)] 

745 # pylint: enable=protected-access 

746 

747 callback(elems) 

748 

749 def find_id(self, elem_id, callback): 

750 def find_id_cb(elems): 

751 """Call the real callback with the found elements.""" 

752 if not elems: 

753 callback(None) 

754 else: 

755 callback(elems[0]) 

756 

757 # Escape non-alphanumeric characters in the selector 

758 # https://www.w3.org/TR/CSS2/syndata.html#value-def-identifier 

759 elem_id = re.sub(r'[^a-zA-Z0-9_-]', r'\\\g<0>', elem_id) 

760 self.find_css('#' + elem_id, find_id_cb, error_cb=lambda exc: None) 

761 

762 def find_focused(self, callback): 

763 frame = cast(QWebFrame | None, self._widget.page().currentFrame()) 

764 if frame is None: 

765 callback(None) 

766 return 

767 

768 elem = frame.findFirstElement('*:focus') 

769 if elem.isNull(): 

770 callback(None) 

771 else: 

772 callback(webkitelem.WebKitElement(elem, tab=self._tab)) 

773 

774 def find_at_pos(self, pos, callback): 

775 assert pos.x() >= 0 

776 assert pos.y() >= 0 

777 frame = cast(QWebFrame | None, self._widget.page().frameAt(pos)) 

778 if frame is None: 

779 # This happens when we click inside the webview, but not actually 

780 # on the QWebPage - for example when clicking the scrollbar 

781 # sometimes. 

782 log.webview.debug("Hit test at {} but frame is None!".format(pos)) 

783 callback(None) 

784 return 

785 

786 # You'd think we have to subtract frame.geometry().topLeft() from the 

787 # position, but it seems QWebFrame::hitTestContent wants a position 

788 # relative to the QWebView, not to the frame. This makes no sense to 

789 # me, but it works this way. 

790 hitresult = frame.hitTestContent(pos) 

791 if hitresult.isNull(): 

792 # For some reason, the whole hit result can be null sometimes (e.g. 

793 # on doodle menu links). 

794 log.webview.debug("Hit test result is null!") 

795 callback(None) 

796 return 

797 

798 try: 

799 elem = webkitelem.WebKitElement(hitresult.element(), tab=self._tab) 

800 except webkitelem.IsNullError: 

801 # For some reason, the hit result element can be a null element 

802 # sometimes (e.g. when clicking the timetable fields on 

803 # https://www.sbb.ch/ ). 

804 log.webview.debug("Hit test result element is null!") 

805 callback(None) 

806 return 

807 

808 callback(elem)
