19class TempDownload(QObject): 

20 

21 """A download of some data into a file object.""" 

22 

23 finished = pyqtSignal() 

24 

25 def __init__(self, item: qtnetworkdownloads.DownloadItem) -> None: 

26 super().__init__() 

27 self._item = item 

28 self._item.finished.connect(self._on_download_finished) 

29 self.successful = False 

30 self.fileobj = item.fileobj 

31 

32 @pyqtSlot() 

33 def _on_download_finished(self) -> None: 

34 self.successful = self._item.successful 

35 self.finished.emit()

38def download_temp(url: QUrl) -> TempDownload: 

39 """Download the given URL into a file object. 

40 

41 The download is not saved to disk. 

42 

43 Returns a ``TempDownload`` object, which triggers a ``finished`` signal 

44 when the download has finished:: 

45 

46 dl = downloads.download_temp(QUrl("https://www.example.com/")) 

47 dl.finished.connect(functools.partial(on_download_finished, dl)) 

48 

49 After the download has finished, its ``successful`` attribute can be 

50 checked to make sure it finished successfully. If so, its contents can be 

51 read by accessing the ``fileobj`` attribute:: 

52 

53 def on_download_finished(download: downloads.TempDownload) -> None: 

54 if download.successful: 

55 print(download.fileobj.read()) 

56 download.fileobj.close() 

57 """ 

58 fobj = io.BytesIO() 

59 fobj.name = 'temporary: ' + url.host() 

60 target = downloads.FileObjDownloadTarget(fobj) 

61 download_manager = objreg.get('qtnetwork-download-manager') 

62 # cache=False is set as a WORKAROUND for MS Defender thinking we're a trojan 

63 # downloader when caching the hostblock list... 

64 return download_manager.get(url, target=target, auto_remove=True, cache=False)
