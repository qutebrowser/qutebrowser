25def early_init(args: argparse.Namespace) -> None: 

26 """Initialize the part of the config which works without a QApplication.""" 

27 configdata.init() 

28 

29 yaml_config = configfiles.YamlConfig() 

30 

31 config.instance = config.Config(yaml_config=yaml_config) 

32 config.val = config.ConfigContainer(config.instance) 

33 configapi.val = config.ConfigContainer(config.instance) 

34 config.key_instance = config.KeyConfig(config.instance) 

35 config.cache = configcache.ConfigCache() 

36 yaml_config.setParent(config.instance) 

37 

38 for cf in config.change_filters: 

39 cf.validate() 

40 

41 config_commands = configcommands.ConfigCommands( 

42 config.instance, config.key_instance) 

43 objreg.register('config-commands', config_commands, command_only=True) 

44 

45 config_file = standarddir.config_py() 

46 custom_config_py = args.config_py is not None 

47 

48 global _init_errors 

49 

50 try: 

51 if os.path.exists(config_file) or custom_config_py: 

52 # If we have a custom --config-py flag, we want it to be fatal if it doesn't 

53 # exist, so we don't silently fall back to autoconfig.yml in that case. 

54 configfiles.read_config_py( 

55 config_file, 

56 warn_autoconfig=not custom_config_py, 

57 ) 

58 else: 

59 configfiles.read_autoconfig() 

60 except configexc.ConfigFileErrors as e: 

61 log.config.error("Error while loading {}".format(e.basename)) 

62 _init_errors = e 

63 

64 try: 

65 configfiles.init() 

66 except configexc.ConfigFileErrors as e: 

67 _init_errors = e 

68 

69 for opt, val in args.temp_settings: 

70 try: 

71 config.instance.set_str(opt, val) 

72 except configexc.Error as e: 

73 message.error("set: {} - {}".format(e.__class__.__name__, e))

83def _update_font_defaults(setting: str) -> None: 

84 """Update all fonts if fonts.default_family/_size was set.""" 

85 if setting not in {'fonts.default_family', 'fonts.default_size'}: 

86 return 

87 

88 configtypes.FontBase.set_defaults(config.val.fonts.default_family, 

89 config.val.fonts.default_size) 

90 

91 for name, opt in configdata.DATA.items(): 

92 if not isinstance(opt.typ, configtypes.FontBase): 

93 continue 

94 

95 value = config.instance.get_obj(name) 

96 if value is None or not (value.endswith(' default_family') or 

97 'default_size ' in value): 

98 continue 

99 

100 config.instance.changed.emit(name)

103def get_backend(args: argparse.Namespace) -> usertypes.Backend: 

104 """Find out what backend to use based on available libraries.""" 

105 str_to_backend = { 

106 'webkit': usertypes.Backend.QtWebKit, 

107 'webengine': usertypes.Backend.QtWebEngine, 

108 } 

109 

110 if args.backend is not None: 

111 return str_to_backend[args.backend] 

112 else: 

113 return str_to_backend[config.val.backend]

116def late_init(save_manager: savemanager.SaveManager) -> None: 

117 """Initialize the rest of the config after the QApplication is created.""" 

118 global _init_errors 

119 if _init_errors is not None:
