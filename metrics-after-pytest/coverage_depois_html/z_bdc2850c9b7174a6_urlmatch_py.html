6 

7See:

9https://cs.chromium.org/chromium/src/extensions/common/url_pattern.cc

27class ParseError(Exception): 

28 

29 """Raised when a pattern could not be parsed.""" 

30 

31 

32class UrlPattern: 

33 

34 """A Chromium-like URL matching pattern. 

35 

36 Class attributes: 

37 _DEFAULT_PORTS: The default ports used for schemes which support ports. 

38 _SCHEMES_WITHOUT_HOST: Schemes which don't need a host. 

39 

40 Attributes: 

41 host: The host to match to, or None for any host. 

42 _pattern: The given pattern as string. 

43 _match_all: Whether the pattern should match all URLs. 

44 _match_subdomains: Whether the pattern should match subdomains of the 

45 given host. 

46 _scheme: The scheme to match to, or None to match any scheme. 

47 Note that with Chromium, '*'/None only matches http/https and 

48 not file/ftp. We deviate from that as per-URL settings aren't 

49 security relevant. 

50 _path: The path to match to, or None for any path. 

51 _port: The port to match to as integer, or None for any port. 

52 """ 

53 

54 _DEFAULT_PORTS = {'https': 443, 'http': 80, 'ftp': 21} 

55 _SCHEMES_WITHOUT_HOST = ['about', 'file', 'data', 'javascript'] 

56 

57 def __init__(self, pattern: str) -> None: 

58 # Make sure all attributes are initialized if we exit early. 

59 self._pattern = pattern 

60 self._match_all = False 

61 self._match_subdomains: bool = False 

62 self._scheme: str | None = None 

63 self.host: str | None = None 

64 self._path: str | None = None 

65 self._port: int | None = None 

66 

67 # > The special pattern <all_urls> matches any URL that starts with a 

68 # > permitted scheme. 

69 if pattern == '<all_urls>': 

70 self._match_all = True 

71 return 

72 

73 if '\0' in pattern: 

74 raise ParseError("May not contain NUL byte") 

75 

76 pattern = self._fixup_pattern(pattern) 

77 

78 # We use urllib.parse instead of QUrl here because it can handle 

79 # hosts with * in them. 

80 try: 

81 parsed = urllib.parse.urlparse(pattern) 

82 except ValueError as e: 

83 raise ParseError(str(e)) 

84 

85 assert parsed is not None 

86 

87 self._init_scheme(parsed) 

88 self._init_host(parsed) 

89 self._init_path(parsed) 

90 self._init_port(parsed) 

91 

92 def _to_tuple(self) -> tuple[ 

93 bool, # _match_all 

94 bool, # _match_subdomains 

95 str | None, # _scheme 

96 str | None, # host 

97 str | None, # _path 

98 int | None, # _port 

99 ]: 

100 """Get a pattern with information used for __eq__/__hash__.""" 

101 return (self._match_all, self._match_subdomains, self._scheme, 

102 self.host, self._path, self._port) 

103 

104 def __hash__(self) -> int: 

105 return hash(self._to_tuple()) 

106 

107 def __eq__(self, other: Any) -> bool: 

108 if not isinstance(other, UrlPattern): 

109 return NotImplemented 

110 return self._to_tuple() == other._to_tuple() 

111 

112 def __repr__(self) -> str: 

113 return utils.get_repr(self, pattern=self._pattern, constructor=True) 

114 

115 def __str__(self) -> str: 

116 return self._pattern

118 def _fixup_pattern(self, pattern: str) -> str: 

119 """Make sure the given pattern is parseable by urllib.parse.""" 

120 if pattern.startswith('*:'): # Any scheme, but *:// is unparsable 

121 pattern = 'any:' + pattern[2:] 

122 

123 schemes = tuple(s + ':' for s in self._SCHEMES_WITHOUT_HOST) 

124 if '://' not in pattern and not pattern.startswith(schemes): 

125 pattern = 'any://' + pattern 

126 

127 # Chromium handles file://foo like file:///foo 

128 # FIXME This doesn't actually strip the hostname correctly. 

129 if (pattern.startswith('file://') and 

130 not pattern.startswith('file:///')): 

131 pattern = 'file:///' + pattern.removeprefix("file://") 

132 

133 return pattern 

134 

135 def _init_scheme(self, parsed: urllib.parse.ParseResult) -> None: 

136 """Parse the scheme from the given URL. 

137 

138 Deviation from Chromium: 

139 - We assume * when no scheme has been given. 

140 """ 

141 if not parsed.scheme: 

142 raise ParseError("Missing scheme") 

143 

144 if parsed.scheme == 'any': 

145 self._scheme = None 

146 return 

147 

148 self._scheme = parsed.scheme 

149 

150 def _init_path(self, parsed: urllib.parse.ParseResult) -> None: 

151 """Parse the path from the given URL. 

152 

153 Deviation from Chromium: 

154 - We assume * when no path has been given. 

155 """ 

156 if self._scheme == 'about' and not parsed.path.strip(): 

157 raise ParseError("Pattern without path") 

158 

159 if parsed.path == '/*': 

160 self._path = None 

161 elif not parsed.path: 

162 # When the user doesn't add a trailing slash, we assume the pattern 

163 # matches any path. 

164 self._path = None 

165 else: 

166 self._path = parsed.path 

167 

168 def _init_host(self, parsed: urllib.parse.ParseResult) -> None: 

169 """Parse the host from the given URL. 

170 

171 Deviation from Chromium: 

172 - http://:1234/ is not a valid URL because it has no host. 

173 - We don't allow patterns for dot/space hosts which QUrl considers 

174 invalid. 

175 """ 

176 if parsed.hostname is None or not parsed.hostname.strip(): 

177 if self._scheme not in self._SCHEMES_WITHOUT_HOST: 

178 raise ParseError("Pattern without host") 

179 assert self.host is None 

180 return 

181 

182 if parsed.netloc.startswith('['): 

183 # Using QUrl parsing to minimize ipv6 addresses 

184 url = QUrl() 

185 url.setHost(parsed.hostname) 

186 if not url.isValid(): 186 ↛ 187line 186 didn't jump to line 187 because the condition on line 186 was never true

187 raise ParseError(url.errorString()) 

188 self.host = url.host() 

189 return 

190 

191 if parsed.hostname == '*': 

192 self._match_subdomains = True 

193 hostname = None 

194 elif parsed.hostname.startswith('*.'): 

195 if len(parsed.hostname) == 2: 

196 # We don't allow just '*.' as a host. 

197 raise ParseError("Pattern without host") 

198 self._match_subdomains = True 

199 hostname = parsed.hostname[2:] 

200 elif set(parsed.hostname) in {frozenset('.'), frozenset('. ')}: 

201 raise ParseError("Invalid host") 

202 else: 

203 hostname = parsed.hostname 

204 

205 if hostname is None: 

206 self.host = None 

207 elif '*' in hostname: 

208 # Only * or *.foo is allowed as host. 

209 raise ParseError("Invalid host wildcard") 

210 else: 

211 self.host = hostname.rstrip('.') 

212 

213 def _init_port(self, parsed: urllib.parse.ParseResult) -> None: 

214 """Parse the port from the given URL. 

215 

216 Deviation from Chromium: 

217 - We use None instead of "*" if there's no port filter. 

218 """ 

219 if parsed.netloc.endswith(':*'): 

220 # We can't access parsed.port as it tries to run int() 

221 self._port = None 

222 elif parsed.netloc.endswith(':'): 

223 raise ParseError("Invalid port: Port is empty") 

224 else: 

225 try: 

226 self._port = parsed.port 

227 except ValueError as e: 

228 raise ParseError("Invalid port: {}".format(e)) 

229 

230 scheme_has_port = (self._scheme in list(self._DEFAULT_PORTS) or 

231 self._scheme is None) 

232 if self._port is not None and not scheme_has_port: 

233 raise ParseError("Ports are unsupported with {} scheme".format( 

234 self._scheme)) 

235 

236 def _matches_scheme(self, scheme: str) -> bool: 

237 return self._scheme is None or self._scheme == scheme 

238 

239 def _matches_host(self, host: str) -> bool: 

240 # FIXME what about multiple dots? 

241 host = host.rstrip('.') 

242 

243 # If we have no host in the match pattern, that means that we're 

244 # matching all hosts, which means we have a match no matter what the 

245 # test host is. 

246 # Contrary to Chromium, we don't need to check for 

247 # self._match_subdomains, as we want to return True here for e.g. 

248 # file:// as well. 

249 if self.host is None: 

250 return True 

251 

252 # If the hosts are exactly equal, we have a match. 

253 if host == self.host: 

254 return True 

255 

256 # Otherwise, we can only match if our match pattern matches subdomains. 

257 if not self._match_subdomains: 

258 return False 

259 

260 # We don't do subdomain matching against IP addresses, so we can give 

261 # up now if the test host is an IP address. 

262 if not utils.raises(ValueError, ipaddress.ip_address, host): 

263 return False 

264 

265 # Check if the test host is a subdomain of our host. 

266 if len(host) <= (len(self.host) + 1): 

267 return False 

268 

269 if not host.endswith(self.host): 

270 return False 

271 

272 return host[len(host) - len(self.host) - 1] == '.' 

273 

274 def _matches_port(self, scheme: str, port: int) -> bool: 

275 if port == -1 and scheme in self._DEFAULT_PORTS: 

276 port = self._DEFAULT_PORTS[scheme] 

277 return self._port is None or self._port == port 

278 

279 def _matches_path(self, path: str) -> bool: 

280 """Match the URL's path. 

281 

282 Deviations from Chromium: 

283 - Chromium only matches <all_urls> with "javascript:" (pathless); but 

284 we also match *://*/* and friends. 

285 """ 

286 if self._path is None: 

287 return True 

288 

289 # Match 'google.com' with 'google.com/' 

290 if path + '/*' == self._path: 

291 return True 

292 

293 # FIXME Chromium seems to have a more optimized glob matching which 

294 # doesn't rely on regexes. Do we need that too? 

295 return fnmatch.fnmatchcase(path, self._path) 

296 

297 def matches(self, qurl: QUrl) -> bool: 

298 """Check if the pattern matches the given QUrl.""" 

299 qtutils.ensure_valid(qurl) 

300 

301 if self._match_all: 

302 return True 

303 

304 if not self._matches_scheme(qurl.scheme()): 

305 return False 

306 # FIXME ignore for file:// like Chromium? 

307 if not self._matches_host(qurl.host()): 

308 return False 

309 if not self._matches_port(qurl.scheme(), qurl.port()): 

310 return False 

311 if not self._matches_path(qurl.path()): 

312 return False 

313 

314 return True
