5"""Implementation of the command debug-cache-stats. 

6 

7Because many modules depend on this command, this needs to have as few 

8dependencies as possible to avoid cyclic dependencies. 

9""" 

10 

11import weakref 

12from typing import Any, TypeVar 

13from collections.abc import MutableMapping, Callable 

14 

15from qutebrowser.utils import log 

16 

17 

18# The callable should be a lru_cache wrapped function 

19_CACHE_FUNCTIONS: MutableMapping[str, Any] = weakref.WeakValueDictionary() 

20 

21 

22_T = TypeVar('_T', bound=Callable[..., Any]) 

23 

24 

25def register(name: str | None = None) -> Callable[[_T], _T]: 

26 """Register a lru_cache wrapped function for debug_cache_stats.""" 

27 def wrapper(fn: _T) -> _T: 

28 fn_name = fn.__name__ if name is None else name 

29 _CACHE_FUNCTIONS[fn_name] = fn 

30 return fn 

31 return wrapper 

32 

33 

34def debug_cache_stats() -> None: 

35 """Print LRU cache stats.""" 

36 for name, fn in _CACHE_FUNCTIONS.items(): 

37 log.misc.info('{}: {}'.format(name, fn.cache_info()))
