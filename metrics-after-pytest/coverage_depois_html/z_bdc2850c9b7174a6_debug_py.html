224def signal_name(sig: pyqtBoundSignal) -> str: 

225 """Get a cleaned up name of a signal. 

226 

227 Unfortunately, the way to get the name of a signal differs based on 

228 bound vs. unbound signals. 

229 

230 Here, we try to get the name from .signal or .signatures, or if all else 

231 fails, extract it from the repr(). 

232 

233 Args: 

234 sig: A bound signal. 

235 

236 Return: 

237 The cleaned up signal name. 

238 """ 

239 if hasattr(sig, 'signal'): 

240 # Bound signal 

241 # Examples: 

242 # sig.signal == '2signal1' 

243 # sig.signal == '2signal2(QString,QString)' 

244 m = re.fullmatch(r'[0-9]+(?P<name>.*)\(.*\)', sig.signal) 

245 else: 

246 # Unbound signal, PyQt >= 5.11 

247 # Examples: 

248 # sig.signatures == ('signal1()',) 

249 # sig.signatures == ('signal2(QString,QString)',) 

250 m = re.fullmatch(r'(?P<name>.*)\(.*\)', 

251 sig.signatures[0]) # type: ignore[attr-defined] 

252 

253 assert m is not None, sig 

254 return m.group('name')
