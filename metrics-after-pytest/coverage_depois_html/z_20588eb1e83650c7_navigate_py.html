23# Order of the segments in a URL. 

24# Each list entry is a tuple of (path name (string), getter, setter). 

25# Note that the getters must not use FullyDecoded decoded mode to prevent loss 

26# of information. (host and path use FullyDecoded by default) 

27_URL_SEGMENTS = [ 

28 ('host', 

29 lambda url: url.host(QUrl.ComponentFormattingOption.FullyEncoded), 

30 lambda url, host: url.setHost(host, QUrl.ParsingMode.StrictMode)), 

31 

32 ('port', 

33 lambda url: str(url.port()) if url.port() > 0 else '', 

34 lambda url, x: url.setPort(int(x))), 

35 

36 ('path', 

37 lambda url: url.path(QUrl.ComponentFormattingOption.FullyEncoded), 

38 lambda url, path: url.setPath(path, QUrl.ParsingMode.StrictMode)), 

39 

40 ('query', 

41 lambda url: url.query(QUrl.ComponentFormattingOption.FullyEncoded), 

42 lambda url, query: url.setQuery(query, QUrl.ParsingMode.StrictMode)), 

43 

44 ('anchor', 

45 lambda url: url.fragment(QUrl.ComponentFormattingOption.FullyEncoded), 

46 lambda url, fragment: url.setFragment(fragment, QUrl.ParsingMode.StrictMode)),

50def _get_incdec_value(match, inc_or_dec, count): 

51 """Get an incremented/decremented URL based on a URL match.""" 

52 pre, zeroes, number, post = match.groups() 

53 # This should always succeed because we match \d+ 

54 val = int(number) 

55 if inc_or_dec == 'decrement': 

56 if val < count: 

57 raise Error("Can't decrement {} by {}!".format(val, count)) 

58 val -= count 

59 elif inc_or_dec == 'increment': 

60 val += count 

61 else: 

62 raise ValueError("Invalid value {} for inc_or_dec!".format(inc_or_dec)) 

63 if zeroes: 

64 if len(number) < len(str(val)): 

65 zeroes = zeroes[1:] 

66 elif len(number) > len(str(val)): 

67 zeroes += '0' 

68 

69 return ''.join([pre, zeroes, str(val), post])

73 """Helper method for :navigate when `where' is increment/decrement. 

74 

75 Args:

77 count: How much to increment or decrement by. 

78 inc_or_dec: Either 'increment' or 'decrement'. 

79 """

88 # Make a copy of the QUrl so we don't modify the original 

89 url = QUrl(url) 

90 # We're searching the last number so we walk the url segments backwards 

91 for segment, getter, setter in reversed(_URL_SEGMENTS): 

92 if segment not in segments: 

93 continue 

94 

95 # Get the last number in a string not preceded by regex '%' or '%.' 

96 match = re.fullmatch(r'(.*\D|^)(?<!%)(?<!%.)(0*)(\d+)(.*)', 

97 getter(url)) 

98 if not match: 

99 continue 

100 

101 setter(url, _get_incdec_value(match, inc_or_dec, count)) 

102 qtutils.ensure_valid(url) 

103 

104 return url

106 raise Error("No number found in URL!")

136def _find_prevnext(prev, elems): 

137 """Find a prev/next element in the given list of elements.""" 

138 # First check for <link rel="prev(ious)|next"> as well as 

139 # e.g. <a class="nav-(prev|next)"> (Hugo) 

140 rel_values = {'prev', 'previous'} if prev else {'next'} 

141 classes = {'nav-prev'} if prev else {'nav-next'} 

142 for e in elems: 

143 if e.tag_name() not in ['link', 'a']: 

144 continue 

145 if 'rel' in e and set(e['rel'].split(' ')) & rel_values: 

146 log.hints.debug("Found {!r} with rel={}".format(e, e['rel'])) 

147 return e 

148 elif e.classes() & classes: 

149 log.hints.debug("Found {!r} with class={}".format(e, e.classes())) 

150 return e 

151 

152 # Then check for regular links/buttons. 

153 elems = [e for e in elems if e.tag_name() != 'link'] 

154 option = 'prev_regexes' if prev else 'next_regexes' 

155 if not elems: 

156 return None 

157 

158 for regex in getattr(config.val.hints, option): 

159 log.hints.vdebug( # type: ignore[attr-defined] 

160 "== Checking regex '{}'.".format(regex.pattern)) 

161 for e in elems: 

162 text = str(e) 

163 if not text: 

164 continue 

165 if regex.search(text): 

166 log.hints.debug("Regex '{}' matched on '{}'.".format( 

167 regex.pattern, text)) 

168 return e 

169 else: 

170 log.hints.vdebug( # type: ignore[attr-defined] 

171 "No match on '{}'!".format(text)) 

172 return None
