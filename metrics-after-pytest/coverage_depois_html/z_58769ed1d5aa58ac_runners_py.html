82def replace_variables(win_id, arglist): 

83 """Utility function to replace variables like {url} in a list of args.""" 

84 tabbed_browser = objreg.get('tabbed-browser', scope='window', 

85 window=win_id) 

86 values: MutableMapping[str, str] = {} 

87 args = [] 

88 

89 def repl_cb(matchobj): 

90 """Return replacement for given match.""" 

91 var = matchobj.group("var") 

92 if var not in values: 

93 values[var] = VARIABLE_REPLACEMENTS[var](tabbed_browser) 

94 return values[var] 

95 

96 try: 

97 for arg in arglist: 

98 # using re.sub with callback function replaces all variables in a 

99 # single pass and avoids expansion of nested variables (e.g. 

100 # "{url}" from clipboard is not expanded) 

101 args.append(VARIABLE_REPLACEMENT_PATTERN.sub(repl_cb, arg)) 

102 except utils.ClipboardError as e: 

103 raise cmdutils.CommandError(e) 

104 return args
