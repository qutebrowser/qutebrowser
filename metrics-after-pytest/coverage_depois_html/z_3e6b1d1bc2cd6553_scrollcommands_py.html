29@cmdutils.register() 

30@cmdutils.argument('tab', value=cmdutils.Value.cur_tab) 

31@cmdutils.argument('count', value=cmdutils.Value.count) 

32def scroll(tab: apitypes.Tab, direction: str, count: int = 1) -> None: 

33 """Scroll the current tab in the given direction. 

34 

35 Note you can use `:cmd-run-with-count` to have a keybinding with a bigger 

36 scroll increment. 

37 

38 Args: 

39 direction: In which direction to scroll 

40 (up/down/left/right/top/bottom). 

41 count: multiplier 

42 """ 

43 # FIXME:mypy Use a callback protocol to enforce having 'count'? 

44 funcs: dict[str, Callable[..., None]] = { 

45 'up': tab.scroller.up, 

46 'down': tab.scroller.down, 

47 'left': tab.scroller.left, 

48 'right': tab.scroller.right, 

49 'top': tab.scroller.top, 

50 'bottom': tab.scroller.bottom, 

51 'page-up': tab.scroller.page_up, 

52 'page-down': tab.scroller.page_down, 

53 } 

54 try: 

55 func = funcs[direction] 

56 except KeyError: 

57 expected_values = ', '.join(sorted(funcs)) 

58 raise cmdutils.CommandError("Invalid value {!r} for direction - " 

59 "expected one of: {}".format( 

60 direction, expected_values)) 

61 

62 if direction in ['top', 'bottom']: 

63 func() 

64 else: 

65 func(count=count)
