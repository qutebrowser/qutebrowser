6"""Early initialization and main entry point. 

7 

8qutebrowser's initialization process roughly looks like this: 

9 

10- This file gets imported, either via the setuptools entry point or 

11 __main__.py. 

12- At import time, we check for the correct Python version and show an error if 

13 it's too old. 

14- The main() function in this file gets invoked 

15- Argument parsing takes place 

16- earlyinit.early_init() gets invoked to do various low-level initialization 

17 and checks whether all dependencies are met. 

18- app.run() gets called, which takes over. 

19 See the docstring of app.py for details. 

20"""

142def logfilter_error(logfilter): 

143 """Validate logger names passed to --logfilter. 

144 

145 Args: 

146 logfilter: A comma separated list of logger names. 

147 """ 

148 from qutebrowser.utils import log

152 raise argparse.ArgumentTypeError(e)

156def debug_flag_error(flag): 

157 """Validate flags passed to --debug-flag. 

158 

159 Available flags: 

160 debug-exit: Turn on debugging of late exit. 

161 pdb-postmortem: Drop into pdb on exceptions. 

162 no-sql-history: Don't store history items. 

163 no-scroll-filtering: Process all scrolling updates. 

164 log-requests: Log all network requests. 

165 log-cookies: Log cookies in cookie filter. 

166 log-scroll-pos: Log all scrolling changes. 

167 log-sensitive-keys: Log keypresses in passthrough modes. 

168 stack: Enable Chromium stack logging. 

169 chromium: Enable Chromium logging. 

170 wait-renderer-process: Wait for debugger in renderer process. 

171 avoid-chromium-init: Enable `--version` without initializing Chromium. 

172 werror: Turn Python warnings into errors. 

173 test-notification-service: Use the testing libnotify service. 

174 caret: Enable debug logging for caret.js. 

175 no-system-pdfjs: Ignore system-wide PDF.js installations 

176 """ 

177 valid_flags = ['debug-exit', 'pdb-postmortem', 'no-sql-history', 

178 'no-scroll-filtering', 'log-requests', 'log-cookies', 

179 'log-scroll-pos', 'log-sensitive-keys', 'stack', 'chromium', 

180 'wait-renderer-process', 'avoid-chromium-init', 'werror', 

181 'test-notification-service', 'log-qt-events', 'caret', 

182 'no-system-pdfjs'] 

183 

184 if flag in valid_flags: 

185 return flag 

186 else: 

187 raise argparse.ArgumentTypeError("Invalid debug flag - valid flags: {}" 

188 .format(', '.join(valid_flags)))

191def _unpack_json_args(args): 

192 """Restore arguments from --json-args after a restart. 

193 

194 When restarting, we serialize the argparse namespace into json, and 

195 construct a "fake" argparse.Namespace here based on the data loaded 

196 from json. 

197 """ 

198 new_args = vars(args) 

199 data = json.loads(args.json_args) 

200 new_args.update(data) 

201 return argparse.Namespace(**new_args)

204def _validate_untrusted_args(argv): 

205 # NOTE: Do not use f-strings here, as this should run with older Python 

206 # versions (so that a proper error can be displayed) 

207 try: 

208 untrusted_idx = argv.index('--untrusted-args') 

209 except ValueError: 

210 return 

211 

212 rest = argv[untrusted_idx + 1:] 

213 if len(rest) > 1: 

214 sys.exit( 

215 "Found multiple arguments ({}) after --untrusted-args, " 

216 "aborting.".format(' '.join(rest))) 

217 

218 for arg in rest: 

219 if arg.startswith(('-', ':')): 

220 sys.exit("Found {} after --untrusted-args, aborting.".format(arg))
